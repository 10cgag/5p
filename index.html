// New logic in index.html const DUMMY = ".........."; // 10 characters hiddenInput.value = DUMMY; hiddenInput.oninput = (e) => { const val = e.target.value; if (val.length < DUMMY.length) { // User is deleting - send as many Backspaces as characters removed const count = DUMMY.length - val.length; for (let i = 0; i < count; i++) { ws.send(JSON.stringify({ type: 'key', value: 'Backspace' })); } } else if (val.length > DUMMY.length) { // User is typing - send the new text const added = val.substring(DUMMY.length); ws.send(JSON.stringify({ type: 'text', value: added })); } // Reset to dummy state immediately hiddenInput.value = DUMMY; };
